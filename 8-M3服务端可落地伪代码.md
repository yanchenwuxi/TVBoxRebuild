## 1. 服务端（Kotlin/JVM 参考，非 Android）

### 1.1 密钥管理（KMS + 会话密钥）
```kotlin
// 服务端持有根密钥（HSM/KMS），客户端永远拿不到
class KeyManager(private val kms: KmsClient) {
    fun rootKeyId(): String = "prod-v1"
    /** ECDH：服务端私钥对客户端公钥派生共享密钥（前向保密） */
    fun sharedKey(clientPub: ECPublicKey): ByteArray {
        val ka = KeyAgreement.getInstance("ECDH"); ka.init(serverPriv)
        ka.doPublic(clientPub); return ka.secret
    }
}
```

### 1.2 `/resolve` 端点（签名校验 + 白名单 + 时效 token + 响应加密）
```kotlin
fun resolve(req: ResolveReq, headers: Map<String,String>): ResolveResp {
    // 1. 时间窗口（±5min 防重放）
    checkWithin(headers["X-Ts"], window = 5*60*1000)

    // 2. nonce 去重（Redis SETNX，TTL 5min）
    check(replayCache.setIfAbsent("nonce:${headers["X-Nonce"]}", "1", ttl = 5*60))

    // 3. HMAC-SHA256 恒定时间比较（防时序攻击）
    val expect = hmacSha256(sessionKey(headers["X-Kid"]),
        req.method + req.path + headers["X-Ts"] + headers["X-Nonce"] + sha256Hex(req.body))
    checkMessageEqual(expect, headers["X-Sign"])

    // 4. 源白名单
    val src = whitelist.get(req.srcId) ?: throw HttpException(404, "源不在白名单")
    check(src.allowDevice(req.devId) && src.enabled)

    // 5. 下发加密 + 时效 token 的播放 URL（不下发明文源）
    val m3u8 = src.fetchEncryptedPlayUrl(req.ep)            // 自有源内容（AES 加密存储）
    val exp = now() + 5*60*1000
    val token = hmacSha256(sessionKey, "$req.srcId:${req.ep}:$exp")
    val aad = "${req.srcId}:${headers["X-Ts"]}".toByteArray()
    val iv = SecureRandom().let { ByteArray(12).also { b -> it.nextBytes(b) } }
    val body = encryptAesGcm(plain = json(m3u8), key = sessionKey, iv = iv, aad = aad)
    return ResolveResp(url = body, iv = iv.hex(), kid = headers["X-Kid"],
                       token = token, expiresAt = exp)
}
```

```kotlin
fun verifyPlay(url: String, token: String, exp: Long): Boolean {
```

## 2. 客户端（Kotlin，挂 OkHttp）

### 2.1 签名拦截器（请求）
```kotlin
class SignInterceptor(
    val sharedKey: (String) -> ByteArray,   // 按 X-Kid 取会话密钥
    val deviceId: () -> String
) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val req = chain.request()
        val ts = System.currentTimeMillis()
        val nonce = SecureRandom().nextInt().toString(16)
        val body = req.body?.bytes() ?: ByteArray(0)
        val sign = hmacSha256Hex(sharedKey("current"),
            "${req.method} ${req.url.encodedPath} $ts $nonce ${sha256Hex(body)}")
        val newReq = req.newBuilder()
            .header("X-Ts", ts.toString())
            .header("X-Nonce", nonce)
            .header("X-Kid", "current")
            .header("X-Sign", sign)
            .header("X-DevId", deviceId())
            .build()
        return chain.proceed(newReq)
    }
}
```

### 2.2 解密拦截器（响应，对应服务端 AES-GCM）
```kotlin
class DecryptInterceptor(val sessionKey: (String) -> ByteArray) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val resp = chain.proceed(chain.request())
        if (resp.header("X-Enc") != "1") return resp      // 明文响应直通
        val iv = resp.header("X-Iv")!!.hexToBytes()
        val aad = "${resp.request.header("X-SrcId")}:${resp.request.header("X-Ts")}".toByteArray()
        val plain = decryptAesGcm(resp.body!!.bytes(), sessionKey(resp.header("X-Kid")!!), iv, aad)
        return resp.newBuilder().body(plain.toResponseBody()).build()
    }
}
```

### 2.3 握手（ECDH，替代某商业闭源影音App conceal 的静态盐）
```kotlin
suspend fun handshake(server: Server): Session {
    val (pub, priv) = HandshakeClient.genKeyPair()
    val resp = server.post("/handshake", pub.hex())       // 下发 AES-GCM 加密的 SessionKey
    val shared = HandshakeClient.deriveSharedKey(priv, resp.clientPub)
    val sessionKey = HandshakeClient.decrypt(resp.encKey, shared, resp.iv, aad = empty())
    return Session(key = sessionKey, kid = "current")
}
```

```kotlin
// 本地文件（QuickJS 源、头像缓存）也加密：
class EncryptedFileIO(private val key: ByteArray) {
    fun write(path: String, plain: ByteArray): String {
        val iv = ByteArray(12).also { SecureRandom().nextBytes(it) }
        val ct = encryptAesGcm(plain, key, iv)
        Files.write(Path.of(path), iv + ct)             // iv || ciphertext
        return iv.hex()
    }
}
```
